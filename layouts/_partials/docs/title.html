{{/* Overrides the theme's docs/title partial. Same title logic, plus a status emoji prefix taken from the page's `status` front matter: not-done -> ❌ pending -> 🟡 approved -> ✅ Sections whose pages carry a status also get "(N out of M)", N = approved. M is the section's `statusTarget` front matter if set, else its page count. Accepts Page as context */}} {{ $title := "" }} {{ if .LinkTitle }} {{ $title = .LinkTitle }} {{ else if .Title }} {{ $title = .Title }} {{ else if and .IsSection .File }} {{ $title = path.Base .File.Dir | humanize | title }} {{ else if and .IsPage .File }} {{ $title = .File.BaseFileName | humanize | title }} {{ else if .IsHome }} {{ $title = .Site.Title }} {{ end }} {{ with .Params.status }} {{ $emoji := index (dict "not-done" "❌" "pending" "🟡" "approved" "✅") . }} {{ if not $emoji }} {{ errorf "%s: unknown status %q, expected not-done, pending or approved" $.Path . }} {{ end }} {{ $title = printf "%s %s" $emoji $title }} {{ end }} {{ if .IsSection }} {{ $tracked := where .RegularPages "Params.status" "in" (slice "not-done" "pending" "approved") }} {{ if $tracked }} {{ $approved := len (where $tracked "Params.status" "approved") }} {{ $total := .Params.statustarget | default (len $tracked) }} {{ $title = printf "%s (%d out of %d)" $title $approved $total }} {{ end }} {{ end }} {{ return $title }}
